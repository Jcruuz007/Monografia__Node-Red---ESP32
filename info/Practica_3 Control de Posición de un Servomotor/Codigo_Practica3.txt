#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>

WebServer server(80);
Servo miServo;
const int servoPin = 23;

const char* ssid = "TU_RED_WIFI";
const char* password = "TU_CONTRASENA";

void setup() {
  Serial.begin(115200);
  miServo.attach(servoPin);
  miServo.write(0); 
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  server.on("/servo", []() {
    if (server.hasArg("angle")) {
      int angulo = server.arg("angle").toInt();
      miServo.write(angulo);
      server.send(200, "text/plain", "OK");
    } else {
      // Manejo de error si la petición llega vacía
      server.send(400, "text/plain", "Falta parametro angle");
    }
  });
  
  server.begin();
}

void loop() {
  server.handleClient();
}
