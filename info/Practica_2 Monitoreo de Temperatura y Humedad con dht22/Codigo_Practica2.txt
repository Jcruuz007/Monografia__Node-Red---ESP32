#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>
#include <DHT.h>

#define DHTPIN 22
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);
WebServer server(80);

const char* ssid = "TU_RED_WIFI";
const char* password = "TU_CONTRASENA";

void setup() {
  Serial.begin(115200);
  dht.begin();
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  Serial.println(WiFi.localIP());
  
  server.on("/datos", []() {
    float temperatura = dht.readTemperature();
    float humedad = dht.readHumidity();
    
    // El operador lógico OR (||) ya está correcto para que el compilador lo lea
    if (isnan(temperatura) || isnan(humedad)) {
      server.send(500, "text/plain", "Error leyendo el sensor");
      return;
    }
    
    String json = "{\"temperatura\": " + String(temperatura) + ", \"humedad\": " + String(humedad) + "}";
    server.send(200, "application/json", json);
  });
  
  server.begin();
}

void loop() {
  server.handleClient();
}
